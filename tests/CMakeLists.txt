function(nexora_add_test name source)
    cmake_parse_arguments(PARSE_ARGV 2 test "KOKKOS" "" "")
    if(test_KOKKOS)
        set(test_main nexora_kokkos_test_main)
    else()
        set(test_main GTest::gtest_main)
    endif()
    add_executable("${name}" "${source}")
    target_link_libraries("${name}" PRIVATE ${test_UNPARSED_ARGUMENTS} ${test_main})
    gtest_discover_tests("${name}" TEST_PREFIX "${name}." DISCOVERY_MODE PRE_TEST
        PROPERTIES TIMEOUT 30 LABELS gtest)
endfunction()
